# Loan Payoff Analysis Model with Interest Rate Sweep
# System Dynamics DSL v1.2

# Time configuration
constant StartTime = 0        # Start at year 0
constant EndTime = 30         # Simulate for 30 years
constant TimeStep = 0.25      # Quarterly calculations

# Initial loan and payment parameters
stock LoanBalance = 300000    # Starting loan amount ($)
constant PaymentAmount = 20000  # Annual payment ($)
constant InterestRate = 0.05    # Annual interest rate (decimal)
constant ExtraPayment = 0       # Additional payment beyond regular payment

# Calculations for loan dynamics
calc AnnualInterest = LoanBalance * InterestRate
calc RemainingPayment = PaymentAmount + ExtraPayment - AnnualInterest

# We'll use a conditional approach instead of a map with variables
# If payment <= 0, pay nothing
# If payment > balance, pay only the balance
# Otherwise pay the calculated amount
calc EffectivePayment = RemainingPayment <= 0 ? 0 : 
                       (RemainingPayment > LoanBalance ? LoanBalance : RemainingPayment)

# Define loan dynamics with flows
flow InterestAccrual:
  AnnualInterest -+> LoanBalance

flow LoanPayment:
  EffectivePayment --> LoanBalance

# Additional metrics
calc PayoffProgress = 1 - (LoanBalance / 300000)
calc YearsToPayoff = time + (LoanBalance / (PaymentAmount + ExtraPayment - AnnualInterest))

# Define a scenario with extra payments
scenario ExtraPayments:
  constant ExtraPayment = 5000

# Parameter sweep to analyze different interest rates
sweep InterestRate = [0.03, 0.04, 0.05, 0.06, 0.07]
