# Economic Growth Model with Environmental Constraints
# Using DSL System Dynamics v1.2 Syntax

# Time configuration
constant StartTime = 2020
constant EndTime = 2100
constant TimeStep = 0.25

# Define modules for the model's subsystems
module Economy:
    # Initial state of economic variables
    stock Capital = 1000
    stock Labor = 100
    stock Technology = 1
    stock Pollution = 10
    
    # Economic model parameters
    constant CapitalDepreciationRate = 0.05
    constant InvestmentFraction = 0.2
    constant LaborGrowthRate = 0.01
    constant TechnologyGrowthRate = 0.01
    constant PollutionPerOutput = 0.1
    constant PollutionAbsorptionTime = 10
    
    # Economic calculations
    calc Output = (Capital ^ 0.75) * (Labor ^ 0.25) * Technology
    calc Investment = Output * InvestmentFraction
    calc Consumption = Output - Investment
    calc ConsumptionPerCapita = Consumption / Population.Population
    calc PollutionGeneration = Output * PollutionPerOutput
    
    # Capital stock dynamics
    flow CapitalFormation:
        Investment -+> Capital
        
    flow CapitalDepreciation:
        Capital * CapitalDepreciationRate --> Capital
    
    # Labor dynamics
    flow LaborGrowth:
        Labor * LaborGrowthRate * Population.FertilityEffect -+> Labor
    
    # Technology dynamics
    flow TechnologyImprovement:
        Technology * TechnologyGrowthRate * ResearchEffectivenessMultiplier -+> Technology
    
    # Pollution dynamics
    flow PollutionCreation:
        PollutionGeneration -+> Pollution
        
    flow PollutionAbsorption:
        Pollution / PollutionAbsorptionTime --> Pollution
    
    # Multiplier accounting for environmental effects on research
    calc ResearchEffectivenessMultiplier = map(ResearchEffectiveness, Environment.EnvironmentalQuality)
    
    # Map defining how environmental quality affects research effectiveness
    map ResearchEffectiveness: spline
        (0.0, 0.5)  # Poor environment: reduced effectiveness
        (0.5, 0.9)  # Moderate environment: slightly reduced effectiveness
        (1.0, 1.0)  # Pristine environment: full effectiveness

module Population:
    # Population stock and parameters
    stock Population = 300
    
    constant BirthRate = 0.03
    constant DeathRate = 0.01
    constant PollutionEffectOnMortality = 0.2
    constant WelfareEffectOnFertility = 0.5
    
    # Population dynamics calculations
    calc MortalityMultiplier = 1 + PollutionEffectOnMortality * Economy.Pollution / 100
    calc ActualDeathRate = DeathRate * MortalityMultiplier
    
    calc WelfareIndex = Economy.ConsumptionPerCapita / 10
    calc FertilityEffect = map(FertilityResponse, WelfareIndex)
    
    # Population flows
    flow Births:
        Population * BirthRate * FertilityEffect -+> Population
        
    flow Deaths:
        Population * ActualDeathRate --> Population
    
    # Map defining how welfare affects fertility
    map FertilityResponse: linear
        (0.0, 1.5)   # Low welfare: high fertility
        (1.0, 1.0)   # Medium welfare: replacement fertility
        (2.0, 0.7)   # High welfare: below replacement fertility
        (3.0, 0.6)   # Very high welfare: low fertility

module Environment:
    # Environmental variables
    stock NaturalResources = 1000
    stock Biodiversity = 100
    
    constant ResourceRegenerationRate = 0.01
    constant MaxResourceCapacity = 1000
    constant ResourceExtractionPerCapital = 0.1
    constant BiodiversityLossRate = 0.02
    
    # Environmental calculations
    calc ResourceExtraction = Economy.Capital * ResourceExtractionPerCapital
    calc ResourceRegenerationPotential = (MaxResourceCapacity - NaturalResources) * ResourceRegenerationRate
    
    calc BiodiversityLossMultiplier = 1 + (Economy.Pollution / 50)
    calc BiodiversityRecoveryRate = 0.01 * (NaturalResources / MaxResourceCapacity)
    
    calc EnvironmentalQuality = (NaturalResources / MaxResourceCapacity + Biodiversity / 100) / 2
    
    # Feedback from environmental quality to economy
    calc ResourceEfficiency = map(EfficiencyResponse, EnvironmentalQuality)
    
    # Natural resources dynamics
    flow ResourceUse:
        ResourceExtraction --> NaturalResources
        
    flow ResourceRegeneration:
        ResourceRegenerationPotential -+> NaturalResources
    
    # Biodiversity dynamics
    flow BiodiversityLoss:
        Biodiversity * BiodiversityLossRate * BiodiversityLossMultiplier --> Biodiversity
        
    flow BiodiversityRecovery:
        Biodiversity * BiodiversityRecoveryRate -+> Biodiversity
    
    # Map defining how environmental quality affects resource efficiency
    map EfficiencyResponse: spline
        (0.0, 0.6)  # Poor environment: inefficient resource use
        (0.5, 0.8)  # Moderate environment: improved efficiency
        (1.0, 1.0)  # Pristine environment: optimal efficiency

# Group for policy parameters that might be adjusted across scenarios
group PolicyParameters:
    constant Economy.InvestmentFraction = 0.2
    constant Economy.PollutionPerOutput = 0.1

# Define alternative scenarios for policy testing
scenario GreenEconomy:
    constant Economy.InvestmentFraction = 0.25
    constant Economy.PollutionPerOutput = 0.05
    constant Economy.TechnologyGrowthRate = 0.015

scenario ResourceDepletion:
    stock Environment.NaturalResources = 500
    constant Environment.ResourceRegenerationRate = 0.005
    constant Economy.PollutionPerOutput = 0.15

scenario RapidPopulationGrowth:
    constant Population.BirthRate = 0.04
    constant Population.DeathRate = 0.008
