# Business Growth Model with Marketing Effectiveness Sweep
# System Dynamics DSL v1.2

# Time configuration
constant StartTime = 0
constant EndTime = 60      # 5 years (60 months)
constant TimeStep = 0.25   # Weekly timestep (approximately)

# Initial business state
stock Customers = 100      # Initial customer base
stock Prospects = 1000     # Initial prospects/leads
stock Revenue = 0          # Cumulative revenue

# Business parameters
constant CustomerSpendPerMonth = 100         # Average revenue per customer per month
constant ProspectConversionRate = 0.05       # Base rate of prospect conversion
constant CustomerChurnRate = 0.02            # Monthly customer churn rate
constant ReferralRate = 0.1                  # Referrals per customer per month
constant MarketingBudgetPercentage = 0.2     # Percentage of revenue spent on marketing
constant MarketingEffectiveness = 0.01       # New prospects per dollar spent on marketing
constant TotalAddressableMarket = 100000     # Total potential customers in the market

# Business calculations
calc MonthlyRevenue = Customers * CustomerSpendPerMonth
calc MarketingBudget = MonthlyRevenue * MarketingBudgetPercentage
calc NewProspectsFromMarketing = MarketingBudget * MarketingEffectiveness
calc RemainingMarketPotential = map(MarketSaturation, (Customers + Prospects) / TotalAddressableMarket)
calc NewProspectsFromReferrals = Customers * ReferralRate * RemainingMarketPotential
calc ConvertedCustomers = Prospects * ProspectConversionRate
calc LostCustomers = Customers * CustomerChurnRate

# Map for market saturation effect (diminishing returns as market saturates)
map MarketSaturation: spline
  (0.0, 1.0)    # At 0% market penetration: full effect
  (0.2, 0.9)    # At 20% market penetration: slight reduction
  (0.5, 0.5)    # At 50% market penetration: half effect
  (0.8, 0.2)    # At 80% market penetration: greatly reduced effect
  (1.0, 0.0)    # At 100% market penetration: no effect (market saturated)

# Define flows for customer acquisition and churn
flow ProspectAcquisition:
  NewProspectsFromMarketing -+> Prospects
  NewProspectsFromReferrals -+> Prospects
  
flow CustomerConversion:
  ConvertedCustomers -+> Customers
  ConvertedCustomers --> Prospects
  
flow CustomerChurn:
  LostCustomers --> Customers
  
flow RevenueGeneration:
  MonthlyRevenue -+> Revenue

# Key performance indicators
calc CustomerLifetimeValue = CustomerSpendPerMonth / CustomerChurnRate
calc CustomerAcquisitionCost = MarketingBudgetPercentage * CustomerSpendPerMonth / ProspectConversionRate
calc MonthsToBreakeven = CustomerAcquisitionCost / CustomerSpendPerMonth
calc ROI = (CustomerLifetimeValue - CustomerAcquisitionCost) / CustomerAcquisitionCost

# Scenarios for different business strategies
scenario HighSpend:
  constant MarketingBudgetPercentage = 0.4
  constant CustomerSpendPerMonth = 80

scenario PremiumOffering:
  constant CustomerSpendPerMonth = 200
  constant ProspectConversionRate = 0.03
  constant CustomerChurnRate = 0.01

# Parameter sweep to analyze different marketing effectiveness values
sweep MarketingEffectiveness = [0.005, 0.01, 0.015, 0.02, 0.025]
