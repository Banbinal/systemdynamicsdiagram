# Population Growth Model with Carrying Capacity
# Using Legacy v0.9.2 DSL Syntax

# Time settings
sim 0 100 0.25

# Stocks (initial values)
stock Population = 100

# Parameters (constants)
param BirthRate = 0.05
param DeathRate = 0.02
param CarryingCapacity = 500

# Auxiliary variables
aux PopulationPressure = Population / CarryingCapacity
aux ActualBirthRate = BirthRate * (1 - PopulationPressure * 0.7)

# Response curve for resource constraint
graph ResourceEffect ((0, 1.2), (0.5, 1.0), (1.0, 0.6), (1.5, 0.3), (2.0, 0.1))

# Flows
flow Births = Population * ActualBirthRate
flow Deaths = Population * DeathRate

# Connect flows to stocks
connect Births -> Population
connect Population <- Deaths
